cmake_minimum_required(VERSION 3.20)
project(grpc_python_plugin LANGUAGES CXX)

if(NOT DEFINED GRPC_SOURCE_DIR)
  message(FATAL_ERROR "GRPC_SOURCE_DIR is required")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ABSL_PROPAGATE_CXX_STD ON)
set(protobuf_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_PROTOC_BINARIES ON CACHE BOOL "" FORCE)
set(protobuf_INSTALL OFF CACHE BOOL "" FORCE)
set(protobuf_WITH_ZLIB OFF CACHE BOOL "" FORCE)

add_subdirectory(
  "${GRPC_SOURCE_DIR}/third_party/protobuf"
  "${CMAKE_CURRENT_BINARY_DIR}/protobuf"
  EXCLUDE_FROM_ALL
)

add_executable(protoc-gen-grpc_python
  "${GRPC_SOURCE_DIR}/src/compiler/python_plugin.cc"
  "${GRPC_SOURCE_DIR}/src/compiler/python_generator.cc"
  "${GRPC_SOURCE_DIR}/src/compiler/proto_parser_helper.cc"
)
target_include_directories(protoc-gen-grpc_python PRIVATE
  "${GRPC_SOURCE_DIR}"
  "${GRPC_SOURCE_DIR}/include"
)
target_link_libraries(protoc-gen-grpc_python PRIVATE
  protobuf::libprotoc
  protobuf::libprotobuf
)
target_link_options(protoc-gen-grpc_python PRIVATE
  -static
  -s
)
